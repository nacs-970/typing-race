    ```ts
    import { genRoomCode, isValidRoomCode, type PlayerId, type RaceState } from "@typing-race/shared";
    import type { Player, Room } from "../race/types.ts";
    import type { WsData } from "../ws/handlers.ts";
    import { broadcastLobbyState, broadcastPlayerLeft } from "../ws/broadcast.ts";

    export const MAX_PLAYERS_PER_ROOM = 8;
    const COLLISION_RETRIES = 3;

    export const rooms = new Map<string, Room>();

    export function createRoom(hostWs: import("bun").ServerWebSocket<WsData>, nickname: string): { code: string; room: Room } {
      let code: string;
      let attempts = 0;
      do {
        code = genRoomCode();
        attempts++;
        if (attempts > COLLISION_RETRIES) throw new Error("room code collision after retries");
      } while (rooms.has(code));

      const room: Room = {
        code,
        hostId: hostWs.data.playerId,
        state: "lobby",
        passageId: null,
        passageText: null,
        startsAtServerMs: null,
        players: new Map(),
        createdAt: Date.now(),
        lastActivityAt: Date.now(),
      };
      rooms.set(code, room);

      addPlayer(code, hostWs.data.playerId, nickname, hostWs, /* isHost = */ true);
      return { code, room };
    }

    export function getRoom(code: string): Room | null {
      return rooms.get(code) ?? null;
    }

    export type AddPlayerResult =
      | { ok: true; player: Player; room: Room }
      | { ok: false; code: "ROOM_NOT_FOUND" | "ROOM_FULL" };

    export function addPlayer(
      code: string,
      playerId: PlayerId,
      nickname: string,
      ws: import("bun").ServerWebSocket<WsData>,
      isHost?: boolean,
    ): AddPlayerResult {
      const room = rooms.get(code);
      if (!room) return { ok: false, code: "ROOM_NOT_FOUND" };
      if (room.players.size >= MAX_PLAYERS_PER_ROOM) return { ok: false, code: "ROOM_FULL" };

      const isFirstPlayer = room.players.size === 0;
      const player: Player = {
        playerId,
        nickname,
        isHost: isHost ?? isFirstPlayer,
        wsRef: ws,
        progress: 0,
        lastKeystrokeAt: 0,
        clientOffsetMs: 0,
        joinedAt: Date.now(),
      };
      room.players.set(playerId, player);
      room.lastActivityAt = Date.now();

      broadcastLobbyState(room);
      return { ok: true, player, room };
    }

    export function removePlayer(code: string, playerId: PlayerId): void {
      const room = rooms.get(code);
      if (!room) return;
      room.players.delete(playerId);
      room.lastActivityAt = Date.now();

      if (room.players.size === 0) {
        rooms.delete(code);
        return;
      }

      // If the host left, promote the next-joined player
      if (room.hostId === playerId) {
        const next = [...room.players.values()].sort((a, b) => a.joinedAt - b.joinedAt)[0];
        if (next) {
          next.isHost = true;
          room.hostId = next.playerId;
        }
      }
      broadcastPlayerLeft(room, playerId);
      broadcastLobbyState(room);
    }
    ```

    ```ts
    import type { RaceState } from "@typing-race/shared";
    import type { Room } from "./types.ts";
    import { broadcastRaceStart, broadcastCountdown } from "../ws/broadcast.ts";
    import { rooms } from "../rooms/manager.ts";

    // FSM transition table — only these are legal
    const TRANSITIONS: Record<RaceState, RaceState[]> = {
      lobby: ["countdown"],
      countdown: ["racing", "lobby"],   // countdown → lobby if player leaves mid-countdown
      racing: ["finished"],
      finished: ["lobby"],              // rematch — Phase 4
    };

    export function transition(room: Room, target: RaceState): boolean {
      if (!TRANSITIONS[room.state].includes(target)) return false;
      room.state = target;
      room.lastActivityAt = Date.now();
      return true;
    }

    export const COUNTDOWN_DURATION_MS = 3000;  // 3s countdown per ROADMAP discretion

    export function startCountdown(room: Room): boolean {
      if (room.state !== "lobby") return false;
      if (room.players.size < 2) return false;  // ≥2 players required per REQ-02
      if (!transition(room, "countdown")) return false;
      room.startsAtServerMs = Date.now() + COUNTDOWN_DURATION_MS;
      broadcastCountdown(room, room.startsAtServerMs, COUNTDOWN_DURATION_MS / 1000);
      return true;
    }

    export function tick(): void {
      const now = Date.now();
      for (const room of rooms.values()) {
        if (room.state !== "countdown") continue;
        if (room.startsAtServerMs === null) continue;
        if (now < room.startsAtServerMs) continue;

        // Time to race — pick a passage (Phase 3 has corpus; for now, use a stub)
        room.passageId = "stub-passage";
        room.passageText = "the quick brown fox jumps over the lazy dog";
        transition(room, "racing");
        broadcastRaceStart(room);
      }
    }
    ```

    ```ts
    import type { Room, Player } from "../race/types.ts";
    import type { ServerToClient } from "@typing-race/shared";

    export function sendToPlayer(player: Player, frame: ServerToClient): void {
      try {
        player.wsRef.send(JSON.stringify(frame));
      } catch {
        // wsRef may be closed; ignore
      }
    }

    export function broadcastToRoom(room: Room, frame: ServerToClient): void {
      for (const player of room.players.values()) sendToPlayer(player, frame);
    }

    export function broadcastLobbyState(room: Room): void {
      broadcastToRoom(room, {
        type: "lobby_state",
        roomCode: room.code,
        players: [...room.players.values()].map((p) => ({
          playerId: p.playerId,
          nickname: p.nickname,
          isHost: p.isHost,
          progress: p.progress,
        })),
      });
    }

    export function broadcastPlayerLeft(room: Room, playerId: string): void {
      broadcastToRoom(room, { type: "player_left", playerId });
    }

    export function broadcastCountdown(room: Room, startsAtServerMs: number, secondsRemaining: number): void {
      broadcastToRoom(room, { type: "countdown", startsAtServerMs, secondsRemaining });
    }

    export function broadcastRaceStart(room: Room): void {
      if (room.startsAtServerMs === null || room.passageText === null || room.passageId === null) return;
      broadcastToRoom(room, {
        type: "race_start",

        passageId: room.passageId,
    ```
