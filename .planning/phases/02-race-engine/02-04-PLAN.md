On acceptance: update `player.progress`, broadcast `cursor_update` to all OTHER players in the room. On rejection: send `error` frame back to the sender only, do NOT broadcast. Purpose: this is the LAST plan in Phase 2 and the only one that delivers the "fair race" core value. Without char-match, a client can type the wrong chars and still get credit. This plan makes the race actually fair.

    import type { Keystroke, ServerErrorCode } from "@typing-race/shared";
    import type { Room, Player } from "./types.ts";

    const PRE_START_GRACE_MS = 50;
    const MIN_INTERVAL_MS = 20;

    export type ValidateResult =
      | { ok: true }
      | { ok: false; reason: ServerErrorCode };

    /**
     * The 4 anti-cheat checks. ALL must pass for a keystroke to be accepted.
     * Server-stamps `player.lastKeystrokeAt` and `player.progress` on success.
     */
    export function validateKeystroke(args: {
      room: Room;
      player: Player;
      frame: Keystroke;
      passageText: string;
      now: number;  // injected for testability
    }): ValidateResult {
      const { room, player, frame, passageText, now } = args;

      // Check #1: server-timestamp on receipt — we use `now`, NOT `frame.clientTs`.
      // This is implicit; no code path uses `frame.clientTs` for timing decisions.

      // Check #2: pre-start reject (state must be racing + grace elapsed)
      if (room.state !== "racing") return { ok: false, reason: "NOT_IN_ROOM" };
      if (room.startsAtServerMs === null) return { ok: false, reason: "NOT_IN_ROOM" };
      if (now < room.startsAtServerMs + PRE_START_GRACE_MS) {
        return { ok: false, reason: "RATE_LIMITED" };
      }

      // Check #3: min-interval (≥20ms between accepted keystrokes)
      if (now - player.lastKeystrokeAt < MIN_INTERVAL_MS) {
        return { ok: false, reason: "RATE_LIMITED" };
      }

      // Check #4: char-match against passage
      if (frame.index < 0 || frame.index >= passageText.length) {
        return { ok: false, reason: "INVALID_FRAME" };
      }
      const expected = passageText[frame.index];
      if (frame.char !== expected) {
        return { ok: false, reason: "INVALID_FRAME" };
      }

      // All checks passed — stamp server state
      player.lastKeystrokeAt = now;
      player.progress = Math.max(player.progress, frame.index + 1);
      return { ok: true };
    }
    ```

    export type Cursor = { playerId: string; index: number; serverTs: number };
    export type CursorState = {
      ownIndex: number;
      cursors: Map<string, { index: number; serverTs: number }>;
    };

    export const useCursorStore = create<CursorState>(() => ({
      ownIndex: 0,
      cursors: new Map(),
    }));

    export function setCursorState(patch: Partial<CursorState>): void {
      useCursorStore.setState(patch);
    }

    let lastCursorSendAt = 0;
    export function shouldSendCursor(now: number): boolean {
      if (now - lastCursorSendAt < 100) return false;  // 10Hz
      lastCursorSendAt = now;
      return true;
    }

    import { useEffect, useRef, useState } from "react";
    import { useCursorStore } from "../store/cursor.ts";

    /**
     * Basic race view — passage render, own cursor (optimistic), opponent
     * cursors (server-confirmed). NO interpolation yet (Phase 5 polish).
     */
    export function RaceView({
      passageText,
      playerId,
      onKeystroke,
    }: {
      passageText: string;
      playerId: string;
      onKeystroke: (index: number, char: string) => void;
    }): React.ReactElement {
      const ownIndex = useCursorStore((s) => s.ownIndex);
      const cursors = useCursorStore((s) => s.cursors);
      const [buffer, setBuffer] = useState("");

      // Listen to physical keypresses; we don't render a textarea — the
      // server is authoritative, so the local "input" is just a buffer of
      // chars the user typed since the last cursor index.
      useEffect(() => {
        const onKey = (ev: KeyboardEvent): void => {
          if (ev.key.length !== 1) return;  // skip modifier keys
          if (ev.ctrlKey || ev.metaKey) return;
          ev.preventDefault();
          const nextIndex = ownIndex + buffer.length;
          if (nextIndex >= passageText.length) return;
          onKeystroke(nextIndex, ev.key);
          setBuffer((b) => b + ev.key);
        };
        window.addEventListener("keydown", onKey);
        return () => window.removeEventListener("keydown", onKey);
      }, [ownIndex, buffer, passageText, onKeystroke]);

      // Server cursor_update for own playerId advances ownIndex
      // (handled in App.tsx via store-bridge)

      // Visual: passage as <span> per char, cursor = a vertical bar <span>
      return (
        <div className="race-view">
          <div className="passage">
            {passageText.split("").map((ch, i) => {
              const isOwnCursor = i === ownIndex + buffer.length;
              const opponentCursors = [...cursors.entries()]
                .filter(([pid, c]) => c.index === i && pid !== playerId)
                .map(([pid]) => pid);
              return (
                <span key={i} className={isOwnCursor ? "char own-cursor" : "char"}>
                  {ch}
                  {opponentCursors.map((pid) => (
                    <span key={pid} className="opponent-cursor" />
                  ))}
                </span>
              );
            })}
          </div>
          <div className="buffer">{buffer}</div>
        </div>
      );
    }
    ```
